# The graphical front-end (Qt 6 Widgets): a library (also linked by the
# GUI tests) and the program. Presentation only: every value is
# computed by the core library.
add_library(primerpicker_gui
  about_dialog.cpp
  analysis_view.cpp
  application_icon.cpp
  colour_scheme.cpp
  criteria_dialog.cpp
  criteria_fields.cpp
  find_enzymes_dialog.cpp
  find_pair_dialog.cpp
  find_pair_view.cpp
  from_molecule_dialog.cpp
  glossary_dialog.cpp
  homology_view.cpp
  main_window.cpp
  molecule_view.cpp
  match_dialog.cpp
  match_view.cpp
  multi_primer_dialog.cpp
  multi_primer_view.cpp
  mutagenize_view.cpp
  pair_dialog.cpp
  pair_view.cpp
  picking_view.cpp
  return_key_filter.cpp
  sequence_dialog.cpp
  sites_view.cpp
  system_defaults_dialog.cpp
  text_table_model.cpp)
set_target_properties(primerpicker_gui PROPERTIES AUTOMOC ON)
# The icon of the windows, built into the library (made by
# packaging/icons/make_icons.sh)
qt_add_resources(primerpicker_gui "icons"
  PREFIX /icons
  BASE "${PROJECT_SOURCE_DIR}/packaging/icons/png"
  FILES
    "${PROJECT_SOURCE_DIR}/packaging/icons/png/primerpicker-small-16.png"
    "${PROJECT_SOURCE_DIR}/packaging/icons/png/primerpicker-small-24.png"
    "${PROJECT_SOURCE_DIR}/packaging/icons/png/primerpicker-small-32.png"
    "${PROJECT_SOURCE_DIR}/packaging/icons/png/primerpicker-small-48.png"
    "${PROJECT_SOURCE_DIR}/packaging/icons/png/primerpicker-64.png"
    "${PROJECT_SOURCE_DIR}/packaging/icons/png/primerpicker-128.png"
    "${PROJECT_SOURCE_DIR}/packaging/icons/png/primerpicker-256.png")
# The example molecule of Help -> Example Molecule (pUC19)
qt_add_resources(primerpicker_gui "examples"
  PREFIX /examples
  BASE "${PROJECT_SOURCE_DIR}/data/molecules"
  FILES "${PROJECT_SOURCE_DIR}/data/molecules/pUC19.gb")
# The licences shown by Help -> About PrimerPicker
qt_add_resources(primerpicker_gui "licences"
  PREFIX /licences
  BASE "${PROJECT_SOURCE_DIR}"
  FILES
    "${PROJECT_SOURCE_DIR}/LICENSE"
    "${PROJECT_SOURCE_DIR}/packaging/licenses/THIRD-PARTY.md"
    "${PROJECT_SOURCE_DIR}/packaging/licenses/LGPL-3.0.txt")
target_link_libraries(primerpicker_gui PUBLIC primerpicker_core Qt6::Widgets Qt6::Concurrent)
target_compile_options(primerpicker_gui PRIVATE ${PRIMERPICKER_WARNING_FLAGS})
# QString conversions and Qt's keyword macros are explicit; no API
# deprecated in the minimum Qt version
target_compile_definitions(primerpicker_gui PUBLIC
  QT_NO_CAST_FROM_ASCII
  QT_NO_CAST_TO_ASCII
  QT_NO_CAST_FROM_BYTEARRAY
  QT_NO_KEYWORDS
  QT_NO_NARROWING_CONVERSIONS_IN_CONNECT
  QT_NO_URL_CAST_FROM_STRING
  QT_DISABLE_DEPRECATED_UP_TO=0x060800)

add_executable(primerpicker-gui WIN32 MACOSX_BUNDLE
  main.cpp)
target_link_libraries(primerpicker-gui PRIVATE primerpicker_gui)
target_compile_options(primerpicker-gui PRIVATE ${PRIMERPICKER_WARNING_FLAGS})
target_compile_definitions(primerpicker-gui PRIVATE PRIMERPICKER_VERSION="${PROJECT_VERSION}")
# The program's icon in the file managers: a Windows resource, the
# icon file of the macOS bundle (packaging/icons/)
if(WIN32)
  enable_language(RC)
  set(PRIMERPICKER_ICO "${PROJECT_SOURCE_DIR}/packaging/icons/primerpicker.ico")
  configure_file("${PROJECT_SOURCE_DIR}/packaging/windows/primerpicker-gui.rc.in" primerpicker-gui.rc @ONLY)
  target_sources(primerpicker-gui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/primerpicker-gui.rc")
endif()
if(APPLE)
  set(primerpicker_icns "${PROJECT_SOURCE_DIR}/packaging/icons/primerpicker.icns")
  set_source_files_properties("${primerpicker_icns}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_sources(primerpicker-gui PRIVATE "${primerpicker_icns}")
endif()
# macOS application bundle (Info.plist generated by CMake); the
# packaged bundle is renamed PrimerPicker.app
set_target_properties(primerpicker-gui PROPERTIES
  MACOSX_BUNDLE_ICON_FILE primerpicker.icns
  MACOSX_BUNDLE_BUNDLE_NAME PrimerPicker
  MACOSX_BUNDLE_GUI_IDENTIFIER io.github.frederic-mahe.primerpicker
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
  MACOSX_BUNDLE_COPYRIGHT "GPL-3.0-or-later")
